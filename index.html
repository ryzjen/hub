Gemini AI Integrace const geminiBtn = document.getElementById('gemini-submit'); const geminiInputEl = document.getElementById('gemini-query'); const geminiOutputEl = document.getElementById('gemini-output'); geminiBtn.addEventListener('click', askGemini); geminiInputEl.addEventListener('keypress', (e) => { if (e.key === 'Enter') askGemini(); }); async function askGemini() { const prompt = geminiInputEl.value.trim(); const apiKey = (localStorage.getItem('hub_gemini_key') || '').trim(); if (!prompt) return; if (!apiKey) { geminiOutputEl.style.display = 'block'; geminiOutputEl.textContent = 'Vlož svůj Gemini API klíč v nastavení (ozubené kolo dole).'; return; } geminiOutputEl.style.display = 'block'; geminiOutputEl.textContent = 'Gemini přemýšlí...'; try { const res = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash:generateContent?key=${apiKey}`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ contents: [{ parts: [{ text: prompt }] }] }) }); const data = await res.json(); if (data.error) { geminiOutputEl.textContent = `Chyba API: ${data.error.message || 'Neplatný klíč'}`; return; } if (data.candidates && data.candidates[0].content && data.candidates[0].content.parts[0].text) { geminiOutputEl.textContent = data.candidates[0].content.parts[0].text; } else { geminiOutputEl.textContent = 'Model nevrátil žádný text.'; } } catch (err) { geminiOutputEl.textContent = 'Chyba připojení k síti nebo blokováno prohlížečem.'; } }
